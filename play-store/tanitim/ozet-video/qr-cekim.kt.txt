@RunWith(RobolectricTestRunner::class)
@GraphicsMode(GraphicsMode.Mode.NATIVE)
@Config(sdk = [34], qualifiers = "tr-w960dp-h540dp-land-television-xhdpi")
class TvQrPromoShotTmp {
    @get:Rule val compose = createAndroidComposeRule<ComponentActivity>()
    private val out = File(System.getenv("CINESTREAM_QR_OUT") ?: "/tmp")
    private val url = "http://192.168.1.35:8765/?t=8kae52pu"

    @Test fun qrDialogAndPhonePage() {
        out.mkdirs()
        TvDevice.overrideForTest = true
        compose.setContent {
            MyApplicationTheme(appTheme = AppTheme.PURE_BLACK, dynamicColor = false) {
                Box(Modifier.fillMaxSize().background(Color(0xFF0E071A))) {
                    LoginScreen(onLoginSuccess = { _, _ -> })
                    Dialog(onDismissRequest = {}) {
                        Card(shape = RoundedCornerShape(24.dp), colors = CardDefaults.cardColors(containerColor = Color(0xFF160E1E)),
                            modifier = Modifier.widthIn(max = 460.dp)) {
                            Column(Modifier.fillMaxWidth().padding(24.dp), horizontalAlignment = Alignment.CenterHorizontally) {
                                Text("Telefonla gir", color = Color.White, fontSize = 20.sp, fontWeight = FontWeight.Bold)
                                Spacer(Modifier.height(6.dp))
                                Text("M3U ya da Xtream bilgilerini telefonundan gönder.", color = Color.White.copy(alpha = 0.7f), fontSize = 13.sp, textAlign = TextAlign.Center)
                                Spacer(Modifier.height(16.dp))
                                Box(Modifier.size(240.dp).clip(RoundedCornerShape(12.dp)).background(Color.White).padding(10.dp)) {
                                    QrCodeImage(content = url, modifier = Modifier.size(220.dp))
                                }
                                Spacer(Modifier.height(8.dp))
                                Text("Aynı Wi-Fi ağında olduğunuza emin olun.", color = Color.White.copy(alpha = 0.6f), fontSize = 12.sp, textAlign = TextAlign.Center)
                                Spacer(Modifier.height(14.dp))
                                Text("1. Telefonunun kamerasıyla QR kodu okut.\n2. Açılan sayfada bilgileri yapıştır, \"TV'ye gönder\"e bas.",
                                    color = Color.White, fontSize = 14.sp, lineHeight = 20.sp)
                                Spacer(Modifier.height(8.dp))
                                Text("QR okutamıyorsan tarayıcıya yaz: $url", color = Color.White.copy(alpha = 0.6f), fontSize = 12.sp,
                                    fontFamily = FontFamily.Monospace, textAlign = TextAlign.Center)
                                Spacer(Modifier.height(14.dp))
                                Row(verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.Center) {
                                    CircularProgressIndicator(Modifier.size(16.dp), strokeWidth = 2.dp, color = Color(0xFFE879F9))
                                    Spacer(Modifier.width(8.dp))
                                    Text("Telefondan bekleniyor…", color = Color.White.copy(alpha = 0.8f), fontSize = 13.sp)
                                }
                                Spacer(Modifier.height(16.dp))
                                TextButton(onClick = {}) { Text("Kapat", color = Color(0xFFE879F9), fontWeight = FontWeight.SemiBold) }
                            }
                        }
                    }
                }
            }
        }
        repeat(20) {
            ShadowLooper.idleMainLooper(100, TimeUnit.MILLISECONDS)
            compose.mainClock.advanceTimeBy(100)
            Thread.sleep(20)
        }
        compose.waitForIdle()
        captureScreenRoboImage(File(out, "tv-qr.png").absolutePath)

        // Telefonda açılan sayfa: gerçek sunucudan alınır.
        val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
        val server = PhoneEntryServer(PhoneEntryMode.PLAYLIST, english = false) { null }
        try {
            check(server.start(scope, "192.168.1.35"))
            val local = server.url()!!.replace("192.168.1.35", "127.0.0.1")
            val html = (URL(local).openConnection() as HttpURLConnection).inputStream.bufferedReader().readText()
            File(out, "telefon-sayfa.html").writeText(html)
        } finally {
            server.stop()
            scope.cancel()
            TvDevice.overrideForTest = null
        }
    }
}
